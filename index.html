Case Study: Automated Order Credit Validation (SAP ERP)

Business Rule: Automatically block Order submission if the Customer's credit status is BLOCKED in SAP ERP, while exposing a real-time LWC UI to Sales Reps.

force-app/main/default/classes/OrderCreditCheckService.cls 
public with sharing class OrderCreditCheckService {
    public static void validateOrderCreditLimits(List<Order> newOrders) {
        Set<Id> accountIds = new Set<Id>();
        for (Order ord : newOrders) {
            if (ord.AccountId != null) accountIds.add(ord.AccountId);
        }
        
        Map<Id, Account> accountMap = new Map<Id, Account>([
            SELECT Id, Description FROM Account WHERE Id IN :accountIds
        ]);
        
        for (Order ord : newOrders) {
            Account acc = accountMap.get(ord.AccountId);
            if (acc != null && acc.Description != null && acc.Description.containsIgnoreCase('BLOCKED')) {
                ord.addError('Order placement failed: Customer account status is BLOCKED in SAP ERP.');
            }
        }
    }
}
